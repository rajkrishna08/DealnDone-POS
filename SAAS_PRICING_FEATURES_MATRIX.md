```bash
TRIAL_ALLOWED_FEATURES=all
```

### **Database Schema**
```sql
-- Users table with plan information
CREATE TABLE users (
    id INTEGER PRIMARY KEY,
    business_name TEXT,
    email TEXT UNIQUE,
    plan_type TEXT DEFAULT 'basic',
    trial_status TEXT DEFAULT 'active',
    trial_ends_at DATETIME,
    trial_days_left INTEGER DEFAULT 14,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- Feature flags table
CREATE TABLE feature_flags (
    name TEXT PRIMARY KEY,
    status TEXT,
    rollout_percentage REAL,
    enabled_plans TEXT,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- Audit log table
CREATE TABLE audit_log (
    id INTEGER PRIMARY KEY,
    user_id INTEGER,
    action TEXT,
    current_plan TEXT,
    required_plan TEXT,
    timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
);
```
